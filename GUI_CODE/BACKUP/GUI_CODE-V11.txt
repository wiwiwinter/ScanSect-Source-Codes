def get_icon_from_svg(svg_data_string, size=QSize(16, 16)):
    renderer = QSvgRenderer(QByteArray(svg_data_string.encode('utf-8')))
    pixmap = QPixmap(size)
    pixmap.fill(Qt.transparent)
    painter = QPainter(pixmap)
    renderer.render(painter)
    painter.end()
    return QIcon(pixmap)

class ModelLoaderWorker(QThread):
    """Background thread for loading 3D models without freezing the UI."""
    finished = Signal(object, str)  # mesh, file_path
    error = Signal(str, str)        # error_message, file_path
    progress = Signal(str)          # status message

    def __init__(self, file_path):
        super().__init__()
        self.file_path = file_path

    def run(self):
        try:
            _ensure_3d_libs()
            self.progress.emit(f"Reading file: {os.path.basename(self.file_path)}...")
            mesh = pv.read(self.file_path)

            if mesh is None or mesh.n_points == 0:
                self.error.emit(
                    f"Loaded mesh is empty or invalid from file:\n{self.file_path}",
                    self.file_path
                )
                return

            # Texture search for OBJ files (done in thread to avoid blocking)
            if self.file_path.lower().endswith('.obj'):
                self.progress.emit("Searching for texture...")
                self._load_obj_texture(mesh)

            self.finished.emit(mesh, self.file_path)

        except Exception as e:
            self.error.emit(str(e), self.file_path)

    def _load_obj_texture(self, mesh):
        """Search and load texture for OBJ files."""
        try:
            mtl_texture_path = self._parse_mtl_file(self.file_path)
            texture_path = self._find_texture_file(self.file_path, mtl_texture_path)

            if texture_path:
                try:
                    texture = pv.read_texture(texture_path)
                    if not hasattr(mesh, 'textures'):
                        mesh.textures = {}
                    mesh.textures['diffuse'] = texture
                    print(f"✓ Texture loaded: {texture_path}")
                except Exception as tex_error:
                    print(f"✗ Error loading texture file: {tex_error}")
            else:
                print(f"✗ No texture file found for {os.path.basename(self.file_path)}")
        except Exception as tex_error:
            print(f"✗ Error during texture search: {tex_error}")

    def _parse_mtl_file(self, obj_file_path):
        mtl_path = obj_file_path.replace('.obj', '.mtl')
        if not os.path.exists(mtl_path):
            return None
        try:
            with open(mtl_path, 'r') as f:
                for line in f:
                    line = line.strip()
                    if line.startswith('map_Kd'):
                        texture_path = line.split(None, 1)[1].strip()
                        return texture_path
        except Exception as e:
            print(f"Error parsing MTL file: {e}")
        return None

    def _find_texture_file(self, obj_file_path, mtl_texture_path=None):
        obj_dir = os.path.dirname(obj_file_path)
        base_name = os.path.splitext(os.path.basename(obj_file_path))[0]
        search_paths = []

        if mtl_texture_path:
            search_paths.append(mtl_texture_path)
            mtl_filename = os.path.basename(mtl_texture_path)
            search_paths.append(os.path.join(obj_dir, mtl_filename))

        common_extensions = ['.png', '.jpg', '.jpeg', '.bmp', '.tiff']
        common_suffixes = ['', '_diffuse', '_u1_v1_diffuse', '_texture', '_color', '_basecolor']

        for suffix in common_suffixes:
            for ext in common_extensions:
                search_paths.append(os.path.join(obj_dir, f"{base_name}{suffix}{ext}"))

        texture_folders = ['textures', 'Textures', 'maps', 'Maps', 'images', 'Images']
        for folder in texture_folders:
            folder_path = os.path.join(obj_dir, folder)
            if os.path.exists(folder_path):
                for suffix in common_suffixes:
                    for ext in common_extensions:
                        search_paths.append(os.path.join(folder_path, f"{base_name}{suffix}{ext}"))

        parent_dir = os.path.dirname(obj_dir)
        if parent_dir and parent_dir != obj_dir:
            for suffix in common_suffixes:
                for ext in common_extensions:
                    search_paths.append(os.path.join(parent_dir, f"{base_name}{suffix}{ext}"))

        for path in search_paths:
            if os.path.exists(path):
                return path
        return None


class FileSaveWorker(QThread):
    """Background thread for file save operations."""
    finished = Signal(str)   # success message
    error = Signal(str)      # error message

    def __init__(self, save_func, *args):
        super().__init__()
        self._save_func = save_func
        self._args = args

    def run(self):
        try:
            result = self._save_func(*self._args)
            self.finished.emit(result or "Saved successfully")
        except Exception as e:
            self.error.emit(str(e))

class MeshManager:
    """Manages 3D mesh state, texture application, and model metadata."""

    def __init__(self):
        self.current_mesh = None
        self.current_path = None
        self.current_actor = None

    @property
    def has_model(self):
        return self.current_mesh is not None

    @property
    def model_basename(self):
        if self.current_path:
            return os.path.basename(self.current_path)
        return "Unknown"

    @property
    def insect_name(self):
        if self.current_path:
            name = os.path.splitext(os.path.basename(self.current_path))[0]
            return "".join(c for c in name if c.isalnum() or c in (' ', '-', '_')).strip()
        return "Unknown"

    def set_mesh(self, mesh, file_path):
        self.current_mesh = mesh
        self.current_path = file_path

    def clear(self):
        self.current_mesh = None
        self.current_path = None
        self.current_actor = None

    def has_textures(self):
        if self.current_mesh is None:
            return False
        if hasattr(self.current_mesh, 'textures') and isinstance(self.current_mesh.textures, dict):
            return any(tex is not None for tex in self.current_mesh.textures.values())
        return False

    def has_uvs(self):
        if self.current_mesh is None:
            return False
        return (hasattr(self.current_mesh, 'active_texture_coordinates')
                and self.current_mesh.active_texture_coordinates is not None)

    def get_first_texture(self):
        if self.has_textures():
            return next(iter(self.current_mesh.textures.values()))
        return None

    def get_metadata_text(self, texture_auto=False, texture_manual=False):
        mesh = self.current_mesh
        if mesh is None:
            return "No model loaded."

        lines = [
            "--- Model Information ---",
            f"File Path: {self.model_basename}",
            f"Number of Points: {mesh.n_points:,}",
            f"Number of Polygons (Cells): {mesh.n_cells:,}",
        ]

        bounds = mesh.bounds
        lines.append("Bounds (X, Y, Z min/max):")
        lines.append(f"  X: {bounds[0]:.2f} to {bounds[1]:.2f}")
        lines.append(f"  Y: {bounds[2]:.2f} to {bounds[3]:.2f}")
        lines.append(f"  Z: {bounds[4]:.2f} to {bounds[5]:.2f}")

        if texture_manual:
            lines.append("Texture Status: Manually Applied")
        elif texture_auto:
            lines.append("Texture Status: Automatically Loaded (e.g., via MTL)")
        elif self.has_textures():
            lines.append("Texture Status: Present in mesh data (likely auto-loaded)")
        else:
            lines.append("Texture Status: None detected or applied")

        if self.has_uvs():
            lines.append("  Texture Coordinates (UVs): Present")
        else:
            lines.append("  Texture Coordinates (UVs): MISSING (Textures cannot be displayed correctly)")

        lines.append("\n--- ScanSect Specific Metadata (Placeholder) ---")
        return "\n".join(lines)

# ArchiveManager - Handles archive I/O
# =====================================================================

class ArchiveManager:
    """Manages archive data persistence and organized folder structure."""

    def __init__(self, archive_dir):
        self.archive_dir = archive_dir
        if not os.path.exists(self.archive_dir):
            os.makedirs(self.archive_dir)
        self.archive_file = os.path.join(self.archive_dir, "archive_metadata.json")
        self.data = self.load()

    def load(self):
        if os.path.exists(self.archive_file):
            try:
                with open(self.archive_file, 'r') as f:
                    return json.load(f)
            except Exception as e:
                print(f"Error loading archive data: {e}")
                return []
        return []

    def save(self):
        try:
            with open(self.archive_file, 'w') as f:
                json.dump(self.data, f, indent=2)
        except Exception as e:
            print(f"Error saving archive data: {e}")
            raise

    def add_entry(self, entry):
        self.data.append(entry)
        self.save()

    def remove_entry(self, entry):
        if entry in self.data:
            self.data.remove(entry)
            self.save()

    def get_organized_path(self, archive_type, insect_name="Unknown"):
        organized_path = os.path.join(self.archive_dir, archive_type, insect_name)
        if not os.path.exists(organized_path):
            os.makedirs(organized_path)
        return organized_path

    def get_unique_insect_names(self):
        names = set()
        for item in self.data:
            model_name = item.get('model_name', 'Unknown')
            if model_name != 'Unknown':
                names.add(model_name)
        return sorted(names)

class LoadingOverlay(QWidget):
    """Opaque overlay shown during model loading.
    
    Must be parented to a regular QWidget container, NOT to a VTK/OpenGL widget
    (QtInteractor), because OpenGL widgets paint over child Qt widgets.
    """

    def __init__(self, parent=None):
        super().__init__(parent)
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setAutoFillBackground(True)
        self.setStyleSheet("background-color: #1a1a1a;")
        self.hide()

        layout = QVBoxLayout(self)
        layout.setAlignment(Qt.AlignCenter)

        self._label = QLabel("Loading model...")
        self._label.setAlignment(Qt.AlignCenter)
        self._label.setStyleSheet("color: #ffffff; font-size: 14pt; font-weight: bold; background: transparent;")
        layout.addWidget(self._label)

        self._progress = QProgressBar()
        self._progress.setRange(0, 0)  # Indeterminate
        self._progress.setFixedWidth(300)
        self._progress.setStyleSheet("""
            QProgressBar { border: 1px solid #555; border-radius: 4px; background: #2c2c2c; height: 20px; }
            QProgressBar::chunk { background-color: #007acc; border-radius: 3px; }
        """)
        layout.addWidget(self._progress, alignment=Qt.AlignCenter)

    def show_message(self, text):
        self._label.setText(text)
        if self.parent():
            self.setGeometry(self.parent().rect())
        self.show()
        self.raise_()

    def hide_overlay(self):
        self.hide()

class ScanSectUnifiedInteractorStyle:
    """Wrapper that creates VTK interactor style lazily (after VTK is loaded)."""

    _instance = None

    @classmethod
    def create(cls):
        _ensure_3d_libs()

        class _Style(vtk.vtkInteractorStyleTrackballCamera):
            def __init__(self):
                super().__init__()
                self.current_left_click_mode = "rotate"

            def set_left_click_mode(self, mode_str):
                self.current_left_click_mode = mode_str.lower()

            def OnLeftButtonDown(self): super().OnLeftButtonDown()
            def OnLeftButtonUp(self): super().OnLeftButtonUp()
            def OnMiddleButtonDown(self): super().OnMiddleButtonDown()
            def OnMiddleButtonUp(self): super().OnMiddleButtonUp()
            def OnRightButtonDown(self): super().OnRightButtonDown()
            def OnRightButtonUp(self): super().OnRightButtonUp()

        return _Style()

    # -----------------------------------------------------------------
    # Event Filter (mouse remapping for pan/zoom)
    # -----------------------------------------------------------------

    def eventFilter(self, obj, event):
        if obj == self.plotter:
            if not isinstance(event, QMouseEvent):
                return super().eventFilter(obj, event)
            current_mode = self.custom_interactor_style.current_left_click_mode

            if event.type() == QEvent.MouseButtonPress and event.button() == Qt.LeftButton:
                target_button = Qt.NoButton
                if current_mode == "pan":
                    target_button = Qt.MiddleButton
                elif current_mode == "zoom":
                    target_button = Qt.RightButton
                if target_button != Qt.NoButton:
                    self._active_remapped_button = target_button
                    remapped = QMouseEvent(event.type(), event.position(), event.globalPosition(),
                                           target_button, event.buttons() | target_button, event.modifiers())
                    QApplication.postEvent(obj, remapped)
                    return True

            elif event.type() == QEvent.MouseMove:
                if event.buttons() & Qt.LeftButton and self._active_remapped_button != Qt.NoButton:
                    remapped_buttons = (event.buttons() & ~Qt.LeftButton) | self._active_remapped_button
                    remapped = QMouseEvent(event.type(), event.position(), event.globalPosition(),
                                           self._active_remapped_button, remapped_buttons, event.modifiers())
                    QApplication.postEvent(obj, remapped)
                    return True

            elif event.type() == QEvent.MouseButtonRelease:
                if event.button() == Qt.LeftButton and self._active_remapped_button != Qt.NoButton:
                    target = self._active_remapped_button
                    self._active_remapped_button = Qt.NoButton
                    remapped_buttons = event.buttons() & ~Qt.LeftButton
                    remapped = QMouseEvent(event.type(), event.position(), event.globalPosition(),
                                           target, remapped_buttons, event.modifiers())
                    QApplication.postEvent(obj, remapped)
                    return True
                elif self._active_remapped_button != Qt.NoButton and event.button() == self._active_remapped_button:
                    self._active_remapped_button = Qt.NoButton

        return super().eventFilter(obj, event)

    @Slot()
    def open_model_dialog(self):
        file_filter = "3D Models (*.ply *.stl *.obj *.vtk *.vtp *.vtu);;Textures (*.png *.jpg *.jpeg *.bmp *.tiff);;All Files (*)"
        start_dir = os.path.dirname(self.mesh_mgr.current_path) if self.mesh_mgr.current_path else os.path.expanduser("~")
        file_path, _ = QFileDialog.getOpenFileName(self, "Open 3D Model or Texture", start_dir, file_filter)
        if file_path:
            if file_path.lower().endswith(('.png', '.jpg', '.jpeg', '.bmp', '.tiff')):
                self.apply_texture_to_current_model(file_path)
            else:
                self.load_model(file_path)

    def load_model(self, file_path):
        """Start loading a model in a background thread."""
        # Prevent concurrent loads
        if self._loader_worker is not None and self._loader_worker.isRunning():
            QMessageBox.warning(self, "Busy", "A model is already being loaded. Please wait.")
            return

        self.loading_overlay.show_message(f"Loading: {os.path.basename(file_path)}...")
        self.status_bar.showMessage(f"Loading model: {file_path}...")

        self._loader_worker = ModelLoaderWorker(file_path)
        self._loader_worker.progress.connect(self._on_load_progress)
        self._loader_worker.finished.connect(self._on_model_loaded)
        self._loader_worker.error.connect(self._on_model_load_error)
        self._loader_worker.start()

    @Slot(str)
    def _on_load_progress(self, message):
        self.loading_overlay.show_message(message)
        self.status_bar.showMessage(message)

    @Slot(object, str)
    def _on_model_loaded(self, mesh, file_path):
        """Called on the main thread when the model has finished loading."""
        self.loading_overlay.hide_overlay()

        try:
            self.mesh_mgr.set_mesh(mesh, file_path)
            self.texture_button.setEnabled(True)

            # Remove old actor
            if self.mesh_mgr.current_actor:
                self.plotter.remove_actor(self.mesh_mgr.current_actor)

            texture_loaded = self.mesh_mgr.has_textures()

            if texture_loaded:
                self.mesh_mgr.current_actor = self.plotter.add_mesh(
                    self.mesh_mgr.current_mesh,
                    texture=self.mesh_mgr.get_first_texture(),
                    smooth_shading=True,
                    show_edges=False,
                    lighting=True,
                    color='white'
                )
            else:
                self.mesh_mgr.current_actor = self.plotter.add_mesh(
                    self.mesh_mgr.current_mesh,
                    smooth_shading=True,
                    show_edges=False,
                    color='white'
                )

            self.plotter.render()
            self.plotter.reset_camera()
            self.plotter.setFocus()

            # Update recent models
            self._add_to_recent_models(file_path)

            mode = self.custom_interactor_style.current_left_click_mode.capitalize()
            self.status_bar.showMessage(
                f"Loaded: {self.mesh_mgr.model_basename}. "
                f"Polys: {mesh.n_cells:,}. Mode: {mode}"
            )
            self.metadata_panel.setText(
                self.mesh_mgr.get_metadata_text(texture_auto=texture_loaded)
            )

            # Show texture info for OBJ files without texture
            if file_path.lower().endswith('.obj') and not texture_loaded:
                QMessageBox.information(
                    self,
                    "No Texture Found",
                    f"Could not find a texture file for this model.\n\n"
                    f"Searched in:\n"
                    f"• Same folder as model\n"
                    f"• Parent folder\n"
                    f"• Common texture subfolders\n"
                    f"• Path from MTL file\n\n"
                    f"You can load a texture manually using:\n"
                    f"'Load Texture' button or File > Open Model/Texture"
                )

        except Exception as e:
            self.status_bar.showMessage(f"Error displaying model: {e}")
            QMessageBox.critical(self, "Model Display Error", f"Could not display model:\n{e}")

    @Slot(str, str)
    def _on_model_load_error(self, error_message, file_path):
        self.loading_overlay.hide_overlay()
        self.status_bar.showMessage(f"Error loading model: {error_message}")
        QMessageBox.critical(self, "Model Load Error", f"Could not load model:\n{error_message}")
        self.mesh_mgr.clear()
        self.texture_button.setEnabled(False)

    # -----------------------------------------------------------------
    # Texture Application
    # -----------------------------------------------------------------

    def apply_texture_to_current_model(self, texture_path):
        if not self.mesh_mgr.has_model:
            QMessageBox.warning(self, "No Model Loaded", "Please load a 3D model before applying a texture.")
            return

        self.status_bar.showMessage(f"Applying texture: {texture_path}...")
        try:
            _ensure_3d_libs()
            texture = pv.read_texture(texture_path)

            if not self.mesh_mgr.has_uvs():
                reply = QMessageBox.question(
                    self, "No UV Coordinates",
                    "This model lacks UV coordinates. Generate basic UV mapping?\n\n"
                    "Note: Results may vary depending on model geometry.",
                    QMessageBox.Yes | QMessageBox.No
                )
                if reply == QMessageBox.Yes:
                    try:
                        self.mesh_mgr.current_mesh.texture_map_to_plane(inplace=True)
                    except Exception as uv_error:
                        QMessageBox.warning(self, "UV Generation Failed",
                                            f"Could not generate UV coordinates:\n{uv_error}")
                else:
                    QMessageBox.warning(self, "Missing UV Coordinates",
                                        "Texture will be applied but may not display correctly without UV coordinates.")

            # Remove existing actor and re-add with texture
            if self.mesh_mgr.current_actor:
                self.plotter.remove_actor(self.mesh_mgr.current_actor)

            self.mesh_mgr.current_actor = self.plotter.add_mesh(
                self.mesh_mgr.current_mesh,
                texture=texture,
                smooth_shading=True,
                show_edges=False,
                lighting=True,
                color='white'
            )

            # Store texture reference
            if not hasattr(self.mesh_mgr.current_mesh, 'textures'):
                self.mesh_mgr.current_mesh.textures = {}
            self.mesh_mgr.current_mesh.textures['diffuse'] = texture

            self.plotter.render()
            self.plotter.reset_camera()
            self.plotter.setFocus()

            self.status_bar.showMessage(f"Texture '{os.path.basename(texture_path)}' applied successfully!")
            self.metadata_panel.setText(self.mesh_mgr.get_metadata_text(texture_manual=True))

        except Exception as e:
            self.status_bar.showMessage(f"Error applying texture: {e}")
            QMessageBox.critical(self, "Texture Error", f"Could not apply texture:\n{e}")

    def load_texture_dialog(self):
        if not self.mesh_mgr.has_model:
            QMessageBox.warning(self, "No Model", "Please load a 3D model first.")

        file_filter = "Texture Files (*.jpg *.jpeg *.png *.bmp *.tiff);;All Files (*)"

        texture_path, _ = QFileDialog.getOpenFileName(self, "Open Texture", start_dir, file_filter)
        if texture_path:
            self.apply_texture_to_current_model(texture_path)

    def toggle_measurement_mode(self):
        _ensure_3d_libs()
        self.is_measuring = self.measure_button.isChecked()

        if self.is_measuring:
            if not hasattr(self.plotter, 'iren') or self.plotter.iren is None:
                QMessageBox.warning(self, "Error", "3D viewer not properly initialized.")
                self.measure_button.setChecked(False)
                return

            try:
                self.line_widget = vtk.vtkLineWidget()
                vtk_interactor = self.plotter.iren.interactor
                self.line_widget.SetInteractor(vtk_interactor)

                if self.mesh_mgr.has_model:
                    bounds = self.mesh_mgr.current_mesh.bounds
                else:
                    bounds = [-1, 1, -1, 1, -1, 1]
                self.line_widget.PlaceWidget(bounds)
                self.line_widget.AddObserver("InteractionEvent", self._update_measurement)

                self.measurement_text_actor = vtk.vtkTextActor()
                self.measurement_text_actor.GetTextProperty().SetFontSize(14)
                self.measurement_text_actor.GetTextProperty().SetColor(1, 1, 1)
                self.measurement_text_actor.SetPosition(10, 10)
                self.plotter.renderer.AddActor(self.measurement_text_actor)

                self.line_widget.SetEnabled(1)
                self._update_measurement(None, None)
                self.save_measurement_button.setEnabled(True)
                self.status_bar.showMessage("Click and drag the line endpoints to measure distance")

            except Exception as e:
                self.measure_button.setChecked(False)
                self.is_measuring = False
                self._cleanup_measurement_widgets()
                QMessageBox.warning(self, "Error", f"Could not initialize measurement tool: {e}")
                return
        else:
            self._cleanup_measurement_widgets()
            self.save_measurement_button.setEnabled(False)
            self.status_bar.clearMessage()

        self.plotter.render()

    def _cleanup_measurement_widgets(self):
        if self.line_widget:
            self.line_widget.SetEnabled(0)
            self.line_widget = None
        if self.measurement_text_actor:
            self.plotter.renderer.RemoveActor(self.measurement_text_actor)
            self.measurement_text_actor = None

    def _update_measurement(self, obj, event):
        if not self.is_measuring or not self.line_widget:
            return
        _ensure_3d_libs()
        point1 = [0, 0, 0]
        point2 = [0, 0, 0]
        self.line_widget.GetPoint1(point1)
        self.line_widget.GetPoint2(point2)

        distance = np.sqrt(sum((p2 - p1) ** 2 for p1, p2 in zip(point1, point2)))

        self.current_measurement = {
            'point1': point1,
            'point2': point2,
            'distance': distance
        }

        measurement_text = f"Distance: {distance:.2f} mm"
        self.measurement_text_actor.SetInput(measurement_text)
        self.status_bar.showMessage(measurement_text)
        self.plotter.render()

    # -----------------------------------------------------------------
    # Screenshot / Save (with background save)
    # -----------------------------------------------------------------

    def create_thumbnail(self, image_path, thumbnail_size=(100, 100)):
        try:
            pixmap = QPixmap(image_path)
            if pixmap.isNull():
                return None
            return pixmap.scaled(thumbnail_size[0], thumbnail_size[1],
                                 Qt.KeepAspectRatio, Qt.SmoothTransformation)
        except Exception:
            return None

    def capture_model_thumbnail(self, save_path=None):
        try:
            if save_path is None:
                timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
                save_path = os.path.join(self.archive_mgr.archive_dir, f"thumb_{timestamp}.png")
            self.plotter.screenshot(save_path, transparent_background=False, window_size=[400, 400])
            return save_path
        except Exception as e:
            print(f"Error capturing thumbnail: {e}")
            return None

    def capture_screenshot(self):
        if not self.mesh_mgr.has_model:
            QMessageBox.warning(self, "No Model", "Please load a 3D model first.")
            return

        description, ok = self._get_archive_description("Screenshot")
        if not ok:
            return

        organized_folder = self.archive_mgr.get_organized_path("Screenshots", self.mesh_mgr.insect_name)
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        filename = f"screenshot_{timestamp}.png"
        file_path = os.path.join(organized_folder, filename)
        thumb_filename = f"thumb_screenshot_{timestamp}.png"
        thumb_path = os.path.join(organized_folder, thumb_filename)

        try:
            # Screenshot capture must happen on main thread (VTK requirement)
            self.plotter.screenshot(file_path)

            thumbnail = self.create_thumbnail(file_path)
            if thumbnail:
                thumbnail.save(thumb_path)

            archive_entry = {
                'timestamp': datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                'type': 'Screenshot',
                'description': description,
                'file_path': file_path,
                'thumbnail_path': thumb_path if thumbnail else None,
                'model_name': self.mesh_mgr.model_basename,
            }
            self.archive_mgr.add_entry(archive_entry)
            self.refresh_archive_list()

            self.status_bar.showMessage(f"Screenshot saved: {filename}")
            QMessageBox.information(self, "Success", f"Screenshot saved successfully!\n{filename}")

        except Exception as e:
            QMessageBox.critical(self, "Capture Error", f"Could not capture screenshot:\n{e}")

    def save_measurement_data(self):
        if not self.is_measuring or not self.current_measurement:
            QMessageBox.warning(self, "No Measurement", "Please create a measurement first.")
            return

        description, ok = self._get_archive_description("Measurement")
        if not ok:
            return

        organized_folder = self.archive_mgr.get_organized_path("Measurements", self.mesh_mgr.insect_name)
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        filename = f"measurement_{timestamp}.txt"
        file_path = os.path.join(organized_folder, filename)
        thumb_filename = f"thumb_measurement_{timestamp}.png"
        thumb_path = os.path.join(organized_folder, thumb_filename)

        try:
            # Capture thumbnail on main thread
            thumb_saved = self.capture_model_thumbnail(thumb_path)

            # Write measurement file
            m = self.current_measurement
            with open(file_path, 'w') as f:
                f.write("ScanSect Measurement Data\n")
                f.write("========================\n\n")
                f.write(f"Timestamp: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")
                f.write(f"Model: {self.mesh_mgr.model_basename}\n")
                f.write(f"Description: {description}\n\n")
                f.write("Measurement:\n")
                f.write(f"  Point 1: ({m['point1'][0]:.4f}, {m['point1'][1]:.4f}, {m['point1'][2]:.4f})\n")
                f.write(f"  Point 2: ({m['point2'][0]:.4f}, {m['point2'][1]:.4f}, {m['point2'][2]:.4f})\n")
                f.write(f"  Distance: {m['distance']:.4f} mm\n")

            archive_entry = {
                'timestamp': datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                'type': 'Measurement',
                'description': description,
                'file_path': file_path,
                'thumbnail_path': thumb_path if thumb_saved else None,
                'model_name': self.mesh_mgr.model_basename,
                'model_path': self.mesh_mgr.current_path,
                'distance': f"{m['distance']:.2f} mm",
                'point1': m['point1'],
                'point2': m['point2'],
            }
            self.archive_mgr.add_entry(archive_entry)
            self.refresh_archive_list()

            self.status_bar.showMessage(f"Measurement saved: {filename}")
            QMessageBox.information(self, "Success", f"Measurement data saved successfully!\n{filename}")

        except Exception as e:
            QMessageBox.critical(self, "Save Error", f"Could not save measurement data:\n{e}")

    def _get_archive_description(self, item_type):
        dialog = QDialog(self)
        dialog.setWindowTitle(f"Save {item_type}")
        dialog.setMinimumWidth(400)

        layout = QVBoxLayout(dialog)
        layout.addWidget(QLabel(f"Enter a description for this {item_type.lower()}:"))

        description_edit = QLineEdit()
        description_edit.setPlaceholderText("E.g., Wing span measurement, Top view, etc.")
        layout.addWidget(description_edit)

        buttons = QDialogButtonBox(QDialogButtonBox.Ok | QDialogButtonBox.Cancel)
        buttons.accepted.connect(dialog.accept)
        buttons.rejected.connect(dialog.reject)
        layout.addWidget(buttons)

        if dialog.exec() == QDialog.Accepted:
            return description_edit.text() or f"{item_type} - {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}", True
        return None, False

    # -----------------------------------------------------------------
    # Archive UI
    # -----------------------------------------------------------------

    def refresh_archive_list(self):
        self.archive_list.clear()
        self.archive_mgr.data = self.archive_mgr.load()

        # Update insect filter
        current_insect = self.archive_insect_filter.currentText()
        self.archive_insect_filter.clear()
        self.archive_insect_filter.addItem("All Insects")
        for insect in self.archive_mgr.get_unique_insect_names():
            self.archive_insect_filter.addItem(insect)
        idx = self.archive_insect_filter.findText(current_insect)
        if idx >= 0:
            self.archive_insect_filter.setCurrentIndex(idx)

        self.apply_archive_filters()

    def apply_archive_filters(self):
        self.archive_list.clear()
        type_filter = self.archive_type_filter.currentText()
        insect_filter = self.archive_insect_filter.currentText()

        for item in reversed(self.archive_mgr.data):
            item_type = item.get('type', 'Unknown')
            if type_filter != "All Types" and item_type != type_filter.rstrip('s'):
                continue

            model_name = item.get('model_name', 'Unknown')
            if insect_filter != "All Insects" and model_name != insect_filter:
                continue

            timestamp = item.get('timestamp', 'Unknown')
            description = item.get('description', 'No description')
            display_text = f"[{timestamp}] {model_name} - {item_type}: {description}"
            list_item = QListWidgetItem(display_text)
            list_item.setData(Qt.UserRole, item)

            # Add thumbnail
            thumbnail_path = item.get('thumbnail_path')
            if thumbnail_path and os.path.exists(thumbnail_path):
                thumb = self.create_thumbnail(thumbnail_path, (80, 80))
                if thumb:
                    list_item.setIcon(QIcon(thumb))
            else:
                file_path = item.get('file_path')
                if file_path and os.path.exists(file_path) and file_path.lower().endswith(('.png', '.jpg', '.jpeg')):
                    thumb = self.create_thumbnail(file_path, (80, 80))
                    if thumb:
                        list_item.setIcon(QIcon(thumb))

            self.archive_list.addItem(list_item)

    def _on_archive_selection_changed(self):
        has_sel = len(self.archive_list.selectedItems()) > 0
        self.delete_archive_button.setEnabled(has_sel)
        self.rename_archive_button.setEnabled(has_sel)

        if has_sel:
            data = self.archive_list.selectedItems()[0].data(Qt.UserRole)
            is_meas = data and data.get('type') == 'Measurement' and 'point1' in data and 'point2' in data
            self.load_measurement_button.setEnabled(is_meas)
        else:
            self.load_measurement_button.setEnabled(False)

    def open_archive_item(self, item):
        data = item.data(Qt.UserRole)
        if data and 'file_path' in data:
            file_path = data['file_path']
            if os.path.exists(file_path):
                os.startfile(file_path)
            else:
                QMessageBox.warning(self, "File Not Found", f"Archive file not found:\n{file_path}")

    def delete_archive_item(self):
        selected_items = self.archive_list.selectedItems()
        if not selected_items:
            return

        reply = QMessageBox.question(self, "Confirm Delete",
                                     f"Delete {len(selected_items)} selected item(s) from archive?",
                                     QMessageBox.Yes | QMessageBox.No)
        if reply != QMessageBox.Yes:
            return

        for item in selected_items:
            data = item.data(Qt.UserRole)
            if data and 'file_path' in data:
                fp = data['file_path']
                if os.path.exists(fp):
                    try:
                        os.remove(fp)
                    except Exception as e:
                        print(f"Error deleting file {fp}: {e}")
            self.archive_mgr.remove_entry(data)

        self.archive_mgr.save()
        self.refresh_archive_list()

    def rename_archive_item(self):
        selected_items = self.archive_list.selectedItems()
        if not selected_items:
            return

        item = selected_items[0]
        data = item.data(Qt.UserRole)
        if not data:
            return

        current_desc = data.get('description', '')
        item_type = data.get('type', 'Item')
        timestamp = data.get('timestamp', 'Unknown')

        dialog = QDialog(self)
        dialog.setWindowTitle(f"Rename {item_type}")
        dialog.setMinimumWidth(450)

        layout = QVBoxLayout(dialog)
        layout.addWidget(QLabel(f"Rename {item_type} from {timestamp}"))
        layout.addWidget(QLabel(f"Current: {current_desc}"))
        layout.addWidget(QLabel("New description:"))

        desc_edit = QLineEdit()
        desc_edit.setText(current_desc)
        desc_edit.setPlaceholderText("Enter new description...")
        desc_edit.selectAll()
        layout.addWidget(desc_edit)

        buttons = QDialogButtonBox(QDialogButtonBox.Ok | QDialogButtonBox.Cancel)
        buttons.accepted.connect(dialog.accept)
        buttons.rejected.connect(dialog.reject)
        layout.addWidget(buttons)

        if dialog.exec() == QDialog.Accepted:
            new_desc = desc_edit.text().strip()
            if not new_desc:
                QMessageBox.warning(self, "Invalid Input", "Description cannot be empty.")
                return
            if new_desc == current_desc:
                return
            try:
                data['description'] = new_desc
                self.archive_mgr.save()
                self.refresh_archive_list()
                # Re-select the renamed item
                for i in range(self.archive_list.count()):
                    li = self.archive_list.item(i)
                    if li.data(Qt.UserRole) == data:
                        self.archive_list.setCurrentItem(li)
                        break
                self.status_bar.showMessage(f"Renamed: '{current_desc}' → '{new_desc}'")
            except Exception as e:
                QMessageBox.critical(self, "Rename Error", f"Could not rename item:\n{e}")

    def open_archive_folder(self):
        if os.path.exists(self.archive_mgr.archive_dir):
            os.startfile(self.archive_mgr.archive_dir)
        else:
            QMessageBox.warning(self, "Folder Not Found", "Archive folder does not exist.")

    def load_measurement_from_archive(self):
        selected_items = self.archive_list.selectedItems()
        if not selected_items:
            return

        data = selected_items[0].data(Qt.UserRole)
        if not data or data.get('type') != 'Measurement':
            QMessageBox.warning(self, "Invalid Selection", "Please select a measurement to load.")
            return

        if 'point1' not in data or 'point2' not in data:
            QMessageBox.warning(self, "Incomplete Data",
                                "This measurement doesn't have position data.\n"
                                "Only measurements saved with the latest version can be reloaded.")
            return

        if not self.mesh_mgr.has_model:
            if 'model_path' in data and os.path.exists(data['model_path']):
                reply = QMessageBox.question(
                    self, "Load Model?",
                    f"No model is currently loaded. Load the associated model?\n\n"
                    f"Model: {data.get('model_name', 'Unknown')}",
                    QMessageBox.Yes | QMessageBox.No
                )
                if reply == QMessageBox.Yes:
                    self.load_model(data['model_path'])
                return
            else:
                QMessageBox.warning(self, "No Model",
                                    "Please load the 3D model this measurement was taken from.")
                return

        try:
            point1 = data['point1']
            point2 = data['point2']

            if not self.is_measuring:
                self.measure_button.setChecked(True)
                self.toggle_measurement_mode()

            if self.line_widget:
                self.line_widget.SetPoint1(point1)
                self.line_widget.SetPoint2(point2)
                self._update_measurement(None, None)
                self.plotter.render()

                desc = data.get('description', 'Loaded measurement')
                dist = data.get('distance', 'Unknown')
                self.status_bar.showMessage(f"Loaded measurement: {desc} ({dist})")
                self.right_panel_tabs.setCurrentIndex(0)

                QMessageBox.information(
                    self, "Measurement Loaded",
                    f"Measurement loaded successfully!\n\n"
                    f"Description: {desc}\nDistance: {dist}\n\n"
                    f"You can now adjust the measurement points or save it again."
                )
            else:
                QMessageBox.warning(self, "Error", "Could not initialize measurement widget.")

        except Exception as e:
            QMessageBox.critical(self, "Load Error", f"Could not load measurement:\n{e}")

    # -----------------------------------------------------------------
    # Recent Models
    # -----------------------------------------------------------------

    def _load_recent_models(self):
        if os.path.exists(self.recent_models_file):
            try:
                with open(self.recent_models_file, 'r') as f:
                    return json.load(f)
            except Exception:
                return []
        return []

    def _save_recent_models(self):
        try:
            with open(self.recent_models_file, 'w') as f:
                json.dump(self.recent_models, f, indent=2)
        except Exception as e:
            print(f"Error saving recent models: {e}")

    def _add_to_recent_models(self, file_path):
        if not os.path.exists(file_path):
            return
        entry = {
            'path': file_path,
            'name': os.path.basename(file_path),
            'directory': os.path.dirname(file_path),
            'last_accessed': datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
            'size': os.path.getsize(file_path),
        }
        self.recent_models = [m for m in self.recent_models if m.get('path') != file_path]
        self.recent_models.insert(0, entry)
        self.recent_models = self.recent_models[:20]
        self._save_recent_models()
        self._refresh_recent_models_list()

    def _refresh_recent_models_list(self):
        self.recent_models_list.clear()
        for model in self.recent_models:
            if not os.path.exists(model.get('path', '')):
                continue
            name = model.get('name', 'Unknown')
            last_accessed = model.get('last_accessed', 'Unknown')
            directory = model.get('directory', '')
            size = model.get('size', 0)
            if size < 1024:
                size_str = f"{size} B"
            elif size < 1024 * 1024:
                size_str = f"{size / 1024:.1f} KB"
            else:
                size_str = f"{size / (1024 * 1024):.1f} MB"
            display_text = f"{name}\n  {size_str} • {last_accessed}\n  {directory}"
            list_item = QListWidgetItem(display_text)
            list_item.setData(Qt.UserRole, model)
            self.recent_models_list.addItem(list_item)

    def _on_recent_selection_changed(self):
        has_sel = len(self.recent_models_list.selectedItems()) > 0
        self.load_recent_button.setEnabled(has_sel)
        self.remove_recent_button.setEnabled(has_sel)

    def _load_recent_model(self, item):
        if item is None:
            return
        data = item.data(Qt.UserRole)
        if data and 'path' in data:
            fp = data['path']
            if os.path.exists(fp):
                self.load_model(fp)
            else:
                QMessageBox.warning(self, "File Not Found",
                                    f"Model file not found:\n{fp}\n\nIt may have been moved or deleted.")
                self.recent_models = [m for m in self.recent_models if m.get('path') != fp]
                self._save_recent_models()
                self._refresh_recent_models_list()

    def _remove_recent_model(self):
        selected_items = self.recent_models_list.selectedItems()
        if not selected_items:
            return
        for item in selected_items:
            data = item.data(Qt.UserRole)
            if data in self.recent_models:
                self.recent_models.remove(data)
        self._save_recent_models()
        self._refresh_recent_models_list()

    def _clear_recent_models(self):
        if not self.recent_models:
            return
        reply = QMessageBox.question(self, "Clear Recent Models",
                                     "Clear all recent models from the list?",
                                     QMessageBox.Yes | QMessageBox.No)
        if reply == QMessageBox.Yes:
            self.recent_models = []
            self._save_recent_models()
            self._refresh_recent_models_list()

    # -----------------------------------------------------------------
    # Cleanup
    # -----------------------------------------------------------------

    def resizeEvent(self, event):
        super().resizeEvent(event)
        # Keep loading overlay sized to viewport
        if hasattr(self, 'loading_overlay') and self.loading_overlay.isVisible():
            self.loading_overlay.setGeometry(self._viewport_container.rect())

    def closeEvent(self, event):
        # Stop any running worker threads
        if self._loader_worker and self._loader_worker.isRunning():
            self._loader_worker.quit()
            self._loader_worker.wait(2000)
        if self._save_worker and self._save_worker.isRunning():
            self._save_worker.quit()
            self._save_worker.wait(2000)

        # Clean up measurement widgets
        self._cleanup_measurement_widgets()

        if hasattr(self, 'plotter') and self.plotter is not None:
            self.plotter.close()
        super().closeEvent(event)


# =====================================================================
# Application Entry Point with Splash Screen
# =====================================================================

def main():
    app = QApplication(sys.argv)
    app.setStyle('Fusion')

    # Splash screen while heavy libs load
    splash_pix = QPixmap(480, 280)
    splash_pix.fill(QColor("#1e1e1e"))
    painter = QPainter(splash_pix)
    painter.setPen(QColor("#007acc"))
    font = QFont("Segoe UI", 28, QFont.Bold)
    painter.setFont(font)
    painter.drawText(splash_pix.rect(), Qt.AlignCenter, "ScanSect")
    painter.setPen(QColor("#888888"))
    font.setPointSize(11)
    font.setBold(False)
    painter.setFont(font)
    painter.drawText(splash_pix.rect().adjusted(0, 60, 0, 0), Qt.AlignCenter, "Loading 3D libraries...")
    painter.end()

    splash = QSplashScreen(splash_pix)
    splash.show()
    app.processEvents()

    # Create main window (this triggers lazy 3D lib loading)
    window = ScanSectGUI()
    window.show()
    splash.finish(window)

    sys.exit(app.exec())


if __name__ == '__main__':
    main()
