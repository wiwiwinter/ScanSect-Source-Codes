"""
ScanSect GUI - Starter Code (Qt Event Filtering for Left-Click Modes)
--------------------------------------------------------------------

This script provides a GUI for the ScanSect project. It uses Qt event
filtering to change the behavior of the LEFT MOUSE BUTTON based on selected
modes (Rotate, Pan, Zoom). Left-clicks are remapped to middle-clicks (for pan)
or right-clicks (for zoom) before reaching the VTK interactor.

Other mouse buttons/wheel retain their standard vtkInteractorStyleTrackballCamera behavior.

It uses PySide6 for the GUI, PyVista for 3D rendering, and VTK.

Changelog:
- Implemented Qt event filtering on the QtInteractor (plotter).
- When in "Pan L-Click" mode, left mouse events are converted to middle mouse events.
- When in "Zoom L-Click" mode, left mouse events are converted to right mouse events.
- ScanSectUnifiedInteractorStyle is simplified, mainly holding the mode.
- Middle mouse, Right mouse, and Scroll Wheel consistently use base class behavior.

Requirements:
- Python 3.7+
- PySide6: pip install PySide6
- PyVista: pip install pyvista
- PyVistaQt (for Qt integration): pip install pyvistaqt
- vtk: (usually installed as a dependency of PyVista) pip install vtk

To Run:
1. Save this code as a Python file (e.g., scanset_gui.py).
2. Make sure you have some 3D model files (e.g., .obj, .ply, .stl) ready.
3. Run from your terminal: python scanset_gui.py
4. Click mode buttons and test left-mouse-button interaction.
"""
import sys
import pyvista as pv
from pyvistaqt import QtInteractor
import vtk 

from PySide6.QtWidgets import (
    QApplication,
    QMainWindow,
    QVBoxLayout,
    QHBoxLayout,
    QWidget,
    QFileDialog,
    QLabel,
    QStatusBar,
    QTextEdit,
    QSizePolicy,
    QPushButton,
    QButtonGroup,
    QAbstractButton
)
from PySide6.QtGui import QAction, QKeySequence, QIcon, QMouseEvent
from PySide6.QtCore import Qt, QEvent, QObject, Slot

# --- Unified Custom Interactor Style (Simplified) ---

class ScanSectUnifiedInteractorStyle(vtk.vtkInteractorStyleTrackballCamera):
    """
    A unified interactor style. Its main role now is to be a placeholder
    and potentially hold mode information if needed by other parts of VTK,
    but the primary interaction remapping is done by a Qt event filter.
    """
    def __init__(self):
        super().__init__()
        # This mode is now primarily for the Qt event filter to query
        self.current_left_click_mode = "rotate" 
        print(f"ScanSectUnifiedInteractorStyle: Initialized. Default left-click mode: '{self.current_left_click_mode}'")

    def set_left_click_mode(self, mode_str):
        old_mode = self.current_left_click_mode
        self.current_left_click_mode = mode_str.lower()
        print(f"ScanSectUnifiedInteractorStyle: Left-click mode changed from '{old_mode}' to '{self.current_left_click_mode}'")

    # Most OnButtonDown/Up methods can now just call super, as the Qt event filter
    # will have already remapped the event if necessary.
    def OnLeftButtonDown(self):
        # print("ScanSectUnifiedInteractorStyle: OnLeftButtonDown (event already processed by Qt filter)")
        super().OnLeftButtonDown()

    def OnLeftButtonUp(self):
        # print("ScanSectUnifiedInteractorStyle: OnLeftButtonUp (event already processed by Qt filter)")
        super().OnLeftButtonUp()
    
    def OnMiddleButtonDown(self):
        # print("ScanSectUnifiedInteractorStyle: OnMiddleButtonDown -> super()")
        super().OnMiddleButtonDown()
    
    def OnMiddleButtonUp(self):
        # print("ScanSectUnifiedInteractorStyle: OnMiddleButtonUp -> super()")
        super().OnMiddleButtonUp()

    def OnRightButtonDown(self):
        # print("ScanSectUnifiedInteractorStyle: OnRightButtonDown -> super()")
        super().OnRightButtonDown()

    def OnRightButtonUp(self):
        # print("ScanSectUnifiedInteractorStyle: OnRightButtonUp -> super()")
        super().OnRightButtonUp()


class ScanSectGUI(QMainWindow):
    """
    Main application window for the ScanSect GUI.
    """

    def __init__(self, parent=None):
        super().__init__(parent)
        self.setWindowTitle("ScanSect - 3D Insect Model Viewer")
        self.setGeometry(100, 100, 1200, 800)

        self.central_widget = QWidget(self)
        self.setCentralWidget(self.central_widget)
        self.main_layout = QHBoxLayout(self.central_widget)

        self.plotter = QtInteractor(self.central_widget)
        
        # --- Interaction Mode Buttons ---
        mode_button_layout = QHBoxLayout()
        self.interaction_mode_group = QButtonGroup(self)
        self.interaction_mode_group.setExclusive(True) 

        self.rotate_button = QPushButton("Rotate L-Click")
        self.rotate_button.setCheckable(True)
        self.rotate_button.setChecked(True) 
        mode_button_layout.addWidget(self.rotate_button)
        self.interaction_mode_group.addButton(self.rotate_button)

        self.pan_button = QPushButton("Pan L-Click")
        self.pan_button.setCheckable(True)
        mode_button_layout.addWidget(self.pan_button)
        self.interaction_mode_group.addButton(self.pan_button)

        self.zoom_button = QPushButton("Zoom L-Click")
        self.zoom_button.setCheckable(True)
        mode_button_layout.addWidget(self.zoom_button)
        self.interaction_mode_group.addButton(self.zoom_button)
        
        left_panel_layout = QVBoxLayout()
        left_panel_layout.addWidget(self.plotter, stretch=1) 
        left_panel_layout.addLayout(mode_button_layout)   

        self.main_layout.addLayout(left_panel_layout, stretch=3)

        # --- Metadata Panel ---
        self.metadata_panel = QTextEdit(self)
        self.metadata_panel.setReadOnly(True)
        self.metadata_panel.setPlaceholderText("Load a model to see its metadata here...")
        self.main_layout.addWidget(self.metadata_panel, stretch=1)

        # --- Plotter Setup ---
        self.plotter.add_axes()
        self.plotter.background_color = 'white'

        # --- Custom Interactor Style and Event Filter ---
        self.custom_interactor_style = ScanSectUnifiedInteractorStyle()
        if hasattr(self.plotter, 'interactor') and self.plotter.interactor is not None:
            self.plotter.interactor.SetInteractorStyle(self.custom_interactor_style)
            print("ScanSectGUI: Unified custom interactor style SET.")
        else:
            print("ScanSectGUI: WARNING - Plotter interactor not available when trying to set custom style.")
        
        # Install event filter on the plotter (the QVTKRenderWindowInteractor widget)
        self.plotter.installEventFilter(self)
        self._active_remapped_button = Qt.NoButton # To track if a left-click was remapped

        self.custom_interactor_style.set_left_click_mode("rotate") 
        self.interaction_mode_group.buttonClicked.connect(self._on_interaction_mode_changed)

        self._create_menus()

        self.status_bar = QStatusBar(self)
        self.setStatusBar(self.status_bar)
        self.status_bar.showMessage("Ready. Left-Click: Rotate. Open a 3D model to begin.")

        self.current_model_path = None
        self.current_model_actor = None

    def eventFilter(self, obj, event):
        # We are only interested in events for self.plotter
        if obj == self.plotter:
            if not isinstance(event, QMouseEvent):
                return super().eventFilter(obj, event) # Pass on non-mouse events

            # Get the current left-click mode from our custom style
            current_mode = self.custom_interactor_style.current_left_click_mode
            
            # --- MouseButtonPress ---
            if event.type() == QEvent.MouseButtonPress:
                if event.button() == Qt.LeftButton:
                    target_button = Qt.NoButton
                    if current_mode == "pan":
                        target_button = Qt.MiddleButton
                        print("EventFilter: LeftClick Press -> Remapping to MiddleClick for Pan")
                    elif current_mode == "zoom":
                        target_button = Qt.RightButton
                        print("EventFilter: LeftClick Press -> Remapping to RightClick for Zoom")
                    
                    if target_button != Qt.NoButton:
                        self._active_remapped_button = target_button # Track the remapped button
                        # Create and post a new event with the remapped button
                        # Note: QMouseEvent constructor parameters might vary slightly across Qt versions
                        # For PySide6, QMouseEvent(type, localPos, windowPos, screenPos, button, buttons, modifiers, source)
                        # We need to ensure globalPos is correctly passed if windowPos is not sufficient
                        remapped_event = QMouseEvent(event.type(), 
                                                     event.position(), # QMouseEvent.position() for localPos
                                                     event.globalPosition(), # QMouseEvent.globalPosition() for globalPos
                                                     target_button, 
                                                     event.buttons() | target_button, # Update buttons state
                                                     event.modifiers())
                        QApplication.postEvent(obj, remapped_event)
                        return True # Event handled, do not pass original left-click
                # For original middle/right clicks, let them pass to VTK
                elif event.button() == Qt.MiddleButton:
                    print(f"EventFilter: MiddleClick Press (Mode: {current_mode}) -> Passing to VTK")
                elif event.button() == Qt.RightButton:
                     print(f"EventFilter: RightClick Press (Mode: {current_mode}) -> Passing to VTK")


            # --- MouseMove ---
            elif event.type() == QEvent.MouseMove:
                # If a left-click was remapped and is currently "pressed" (simulated)
                if event.buttons() & Qt.LeftButton and self._active_remapped_button != Qt.NoButton:
                    print(f"EventFilter: LeftButton Move while remapped to {self._active_remapped_button} -> Remapping Move")
                    # Create and post a new move event with the remapped button state
                    # Ensure the 'buttons' argument reflects the remapped button being pressed
                    remapped_buttons_state = (event.buttons() & ~Qt.LeftButton) | self._active_remapped_button
                    remapped_event = QMouseEvent(event.type(), 
                                                 event.position(),
                                                 event.globalPosition(),
                                                 self._active_remapped_button, # This indicates which button caused the event (not strictly necessary for move)
                                                 remapped_buttons_state,       # This is the crucial part for VTK
                                                 event.modifiers())
                    QApplication.postEvent(obj, remapped_event)
                    return True # Event handled

            # --- MouseButtonRelease ---
            elif event.type() == QEvent.MouseButtonRelease:
                if event.button() == Qt.LeftButton and self._active_remapped_button != Qt.NoButton:
                    print(f"EventFilter: LeftClick Release while remapped to {self._active_remapped_button} -> Remapping Release")
                    # Create and post a new release event with the remapped button
                    target_button_for_release = self._active_remapped_button
                    self._active_remapped_button = Qt.NoButton # Reset remapped state
                    
                    # For release, buttons state should not include the released button
                    remapped_buttons_state = event.buttons() & ~Qt.LeftButton 
                    remapped_event = QMouseEvent(event.type(), 
                                                 event.position(),
                                                 event.globalPosition(),
                                                 target_button_for_release, # The button that was released
                                                 remapped_buttons_state,    # Updated overall buttons state
                                                 event.modifiers())
                    QApplication.postEvent(obj, remapped_event)
                    return True # Event handled
                elif self._active_remapped_button != Qt.NoButton and event.button() == self._active_remapped_button:
                    # This case handles if somehow the original remapped button release comes through
                    # Though less likely if we consume the left button release.
                    print(f"EventFilter: Original remapped button {event.button()} release -> Resetting active remapped button.")
                    self._active_remapped_button = Qt.NoButton
                # For original middle/right releases, let them pass
                elif event.button() == Qt.MiddleButton:
                    print(f"EventFilter: MiddleClick Release (Mode: {current_mode}) -> Passing to VTK")
                elif event.button() == Qt.RightButton:
                     print(f"EventFilter: RightClick Release (Mode: {current_mode}) -> Passing to VTK")


        # For all other events or objects, use the default filter
        return super().eventFilter(obj, event)

    def _create_menus(self):
        menu_bar = self.menuBar()
        file_menu = menu_bar.addMenu("&File")
        open_action = QAction("&Open Model...", self)
        open_action.setShortcut(QKeySequence.Open)
        open_action.triggered.connect(self.open_model_dialog)
        file_menu.addAction(open_action)
        file_menu.addSeparator()
        exit_action = QAction("&Exit", self)
        exit_action.setShortcut(QKeySequence.Quit)
        exit_action.triggered.connect(self.close)
        file_menu.addAction(exit_action)

    @Slot(QAbstractButton) 
    def _on_interaction_mode_changed(self, button_clicked):
        mode_string = "rotate" 
        if button_clicked == self.rotate_button: mode_string = "rotate"
        elif button_clicked == self.pan_button: mode_string = "pan"
        elif button_clicked == self.zoom_button: mode_string = "zoom"
        
        print(f"ScanSectGUI: Button '{button_clicked.text()}' clicked. Setting left-click mode to: '{mode_string}' for event filter.")
        self.custom_interactor_style.set_left_click_mode(mode_string) # Inform the style (and thus filter)
        self.status_bar.showMessage(f"Left-Click Mode: {mode_string.capitalize()}")
        self.plotter.setFocus() 

    @Slot()
    def open_model_dialog(self):
        file_filter = "3D Models (*.ply *.stl *.obj *.vtk *.vtp *.vtu);;All Files (*)"
        file_path, _ = QFileDialog.getOpenFileName(self, "Open 3D Model", "", file_filter)
        if file_path:
            self.load_model(file_path)

    def load_model(self, file_path):
        self.status_bar.showMessage(f"Loading model: {file_path}...")
        QApplication.processEvents() 
        try:
            if self.current_model_actor:
                self.plotter.remove_actor(self.current_model_actor)
            mesh = pv.read(file_path)
            self.current_model_actor = self.plotter.add_mesh(mesh, color="lightgrey", smooth_shading=True)
            self.plotter.reset_camera()
            self.plotter.setFocus() 
            self.current_model_path = file_path
            
            current_left_click_mode = "Unknown"
            if self.custom_interactor_style:
                current_left_click_mode = self.custom_interactor_style.current_left_click_mode.capitalize()
            self.status_bar.showMessage(f"Successfully loaded: {file_path}. Left-Click Mode: {current_left_click_mode}")
            self.display_metadata(mesh, file_path)
        except Exception as e:
            self.status_bar.showMessage(f"Error loading model: {e}")
            print(f"ScanSectGUI: Error loading model '{file_path}': {e}")
            self.metadata_panel.setText(f"Could not load model: {file_path}\nError: {e}")

    def display_metadata(self, mesh, file_path):
        metadata_text = f"--- Model Information ---\n"
        metadata_text += f"File Path: {file_path}\n"
        metadata_text += f"Number of Points: {mesh.n_points}\n"
        metadata_text += f"Number of Cells: {mesh.n_cells}\n"
        bounds = mesh.bounds
        metadata_text += f"Bounds (X, Y, Z min/max):\n"
        metadata_text += f"  X: {bounds[0]:.2f} to {bounds[1]:.2f}\n"
        metadata_text += f"  Y: {bounds[2]:.2f} to {bounds[3]:.2f}\n"
        metadata_text += f"  Z: {bounds[4]:.2f} to {bounds[5]:.2f}\n"
        metadata_text += f"\n--- ScanSect Specific Metadata (Placeholder) ---\n"
        self.metadata_panel.setText(metadata_text)

    def closeEvent(self, event):
        print("ScanSectGUI: Closing application.")
        self.plotter.close() 
        super().closeEvent(event)

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = ScanSectGUI()
    window.show()
    sys.exit(app.exec())
